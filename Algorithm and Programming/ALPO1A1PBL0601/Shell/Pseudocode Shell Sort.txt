procedure ShellSort(A, n)
    gap ← n / 2                      (integer division)
    while gap > 0
        for i ← gap to n - 1
            temp ← A[i]
            j ← i
            while j >= gap and A[j - gap] > temp
                A[j] ← A[j - gap]
                j ← j - gap
            A[j] ← temp
        gap ← gap / 2
