procedure QuickSort(A, low, high)
    if low < high
        p ← Partition(A, low, high)
        QuickSort(A, low, p - 1)
        QuickSort(A, p + 1, high)

procedure Partition(A, low, high)
    pivot ← A[high]
    i ← low - 1
    for j ← low to high - 1
        if A[j] <= pivot
            i ← i + 1
            swap A[i], A[j]
    swap A[i + 1], A[high]
    return i + 1
